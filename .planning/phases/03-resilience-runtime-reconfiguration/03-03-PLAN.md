# 03-03 PLAN: Error Code Mapping + Tests

## Goal

Map `CaptureResult.ErrorCode` values to correct HTTP status codes in `CaptureHandler`, include `VendorErrorCode` and `Timestamp` in error responses, add test doubles for MockScannerAdapter, and add unit/integration tests for backoff, config reload, and error flows.

## Scope

D-10 (HTTP status mapping), D-11 (VendorErrorCode in response), D-12 (per-adapter error translation), D-13 (timeout enforcement), D-14 (in-flight fail), plus unit and integration tests.

## Pre-Existing Compilation Fix Applied

**CRITICAL**: `CaptureResult.ErrorCode` property was missing. The `Fail(string errorCode, string message)` factory method accepted an error code parameter but never stored it. This has been fixed directly in `CaptureResult.cs` — `ErrorCode` property added and `Fail()` now correctly populates it. ScannerManager's existing `Fail()` calls with two arguments now compile and work correctly.

All `Fail()` calls in ScannerManager (lines 182, 221, 225) that were previously passing two arguments to a method that only accepted one parameter now work correctly after this fix.

    [TestClass]
    public class ScannerManagerBackoffTests
    {
        [TestMethod]
        public void BackoffStep_StartsAtZero()
        {
            var mock = new MockScannerAdapterWithSettableProperties { IsConnectedValue = true, InitializeResult = true, ScanResult = CaptureResult.Fail("CAPTURE_FAILED", "fail") };
            var manager = new ScannerManager(new[] { mock }, null);

            Assert.AreEqual(0, manager.BackoffStep);
            Assert.IsFalse(manager.InBackoff);
        }

        [TestMethod]
        public void BackoffStep_IncrementsAfterAllAdapterFailure()
        {
            var failing = new MockScannerAdapterWithSettableProperties { IsConnectedValue = false, InitializeResult = false };
            var manager = new ScannerManager(new[] { failing }, null);

            // First call: all fail, step becomes 1
            manager.Scan();
            Assert.AreEqual(1, manager.BackoffStep);

            // Second call: all fail again, step becomes 2
            manager.Scan();
            Assert.AreEqual(2, manager.BackoffStep);
        }

        [TestMethod]
        public void BackoffStep_CapsAtThree()
        {
            var failing = new MockScannerAdapterWithSettableProperties { IsConnectedValue = false, InitializeResult = false };
            var manager = new ScannerManager(new[] { failing }, null);

            // Trigger many failures
            for (int i = 0; i &lt; 10; i++)
                manager.Scan();

            Assert.AreEqual(3, manager.BackoffStep);
        }

        [TestMethod]
        public void BackoffStep_ResetsOnSuccessfulCapture()
        {
            var failing = new MockScannerAdapterWithSettableProperties { IsConnectedValue = false, InitializeResult = false };
            var succeeding = new MockScannerAdapterWithSettableProperties { IsConnectedValue = true, InitializeResult = true, ScanResult = CaptureResult.Ok(new byte[] { 1, 2, 3 }) };
            var manager = new ScannerManager(new[] { failing, succeeding }, null);

            manager.Scan(); // fail
            manager.Scan(); // fail
            Assert.AreEqual(2, manager.BackoffStep);

            manager.Scan(); // success — reset
            Assert.AreEqual(0, manager.BackoffStep);
        }
    }

    Uses MockScannerAdapterWithSettableProperties from MockScannerAdapterTestDoubles.cs.

    [TestClass]
    public class ErrorHandlingTests
    {
        private HttpListener _listener;
        private Thread _serverThread;
        private HttpListenerContext _capturedContext;
        private readonly ManualResetEvent _contextReady = new ManualResetEvent(false);

        [TestInitialize]
        public void Setup()
        {
            _listener = new HttpListener();
            _listener.Prefixes.Add("http://localhost:0/");
            _listener.Start();
            _serverThread = new Thread(() =>
            {
                _capturedContext = _listener.GetContext();
                _contextReady.Set();
                // Keep server alive while test runs
                _listener.Stop();
            });
            _serverThread.IsBackground = true;
            _serverThread.Start();
        }

        [TestCleanup]
        public void Teardown()
        {
            try { _listener?.Close(); } catch { }
            _contextReady.Dispose();
        }

        [TestMethod]
        public async Task CaptureHandler_Returns503_WhenScannerNotConnected()
        {
            var mock = new MockScannerAdapterWithSettableProperties
            {
                IsConnectedValue = false,
                InitializeResult = false,
                VendorErrorCodeValue = "NO_DEVICE"
            };
            var handler = new CaptureHandler(null);

            // Set up the request path
            _contextReady.Reset();
            var url = _listener.Prefixes[0] + "api/capture";

            // Make async request in parallel
            var responseTask = Task.Run(() =>
            {
                var request = WebRequest.CreateHttp(url);
                request.Method = "POST";
                request.ContentType = "application/json";
                var body = "{\"thamChieuId\":\"test\",\"maPhieu\":\"P001\"}";
                using (var rs = request.GetRequestStream())
                using (var sw = new StreamWriter(rs))
                    sw.Write(body);
                return request.GetResponse();
            });

            // Wait for context, handle it, respond
            await Task.Run(() => _contextReady.WaitOne(5000));
            var ctx = _listener.GetContext();
            handler.Handle(ctx, mock);

            var response = (HttpWebResponse)responseTask;
            Assert.AreEqual(503, (int)response.StatusCode);
        }

        // Similar for CAPTURE_TIMEOUT (504), CAPTURE_FAILED (500), etc.
    }
