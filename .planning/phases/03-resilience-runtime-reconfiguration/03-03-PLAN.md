    [TestClass]
    public class ScannerManagerBackoffTests
    {
        [TestMethod]
        public void BackoffStep_StartsAtZero()
        {
            var mock = new MockScannerAdapter { IsConnectedValue = true, InitializeResult = true, ScanResult = CaptureResult.Fail("CAPTURE_FAILED", "fail") };
            var manager = new ScannerManager(new[] { mock }, null);

            Assert.AreEqual(0, manager.BackoffStep);
            Assert.IsFalse(manager.InBackoff);
        }

        [TestMethod]
        public void BackoffStep_IncrementsAfterAllAdapterFailure()
        {
            var failing = new MockScannerAdapter { IsConnectedValue = false, InitializeResult = false };
            var manager = new ScannerManager(new[] { failing }, null);

            // First call: all fail, step becomes 1
            manager.Scan();
            Assert.AreEqual(1, manager.BackoffStep);

            // Second call: all fail again, step becomes 2
            manager.Scan();
            Assert.AreEqual(2, manager.BackoffStep);
        }

        [TestMethod]
        public void BackoffStep_CapsAtThree()
        {
            var failing = new MockScannerAdapter { IsConnectedValue = false, InitializeResult = false };
            var manager = new ScannerManager(new[] { failing }, null);

            // Trigger many failures
            for (int i = 0; i < 10; i++)
                manager.Scan();

            Assert.AreEqual(3, manager.BackoffStep);
        }

        [TestMethod]
        public void BackoffStep_ResetsOnSuccessfulCapture()
        {
            var failing = new MockScannerAdapter { IsConnectedValue = false, InitializeResult = false };
            var succeeding = new MockScannerAdapter { IsConnectedValue = true, InitializeResult = true, ScanResult = CaptureResult.Ok(new byte[] { 1, 2, 3 }) };
            var manager = new ScannerManager(new[] { failing, succeeding }, null);

            manager.Scan(); // fail
            manager.Scan(); // fail
            Assert.AreEqual(2, manager.BackoffStep);

            manager.Scan(); // success — reset
            Assert.AreEqual(0, manager.BackoffStep);
        }
    }

    Note: MockScannerAdapter needs settable properties (IsConnectedValue, InitializeResult, ScanResult) for test injection.
    If MockScannerAdapter doesn't currently support these as settable properties, add internal setters or a test constructor
    to MockScannerAdapter as part of this task. Check the existing MockScannerAdapter implementation first.
  </description>
</task>
